📖 OPERACIONES DISPONIBLES
═════════════════════════════════════════════════════════════════════════════════

CategoriaDAO dao = new CategoriaDAO();

1️⃣ CREAR TABLA
   dao.createTable();

2️⃣ AGREGAR (CREATE/INSERT)
   int id = dao.add("Descripción de la categoría");

   Ejemplo:
   int id1 = dao.add("Electrónica");
   int id2 = dao.add("Ropa");

3️⃣ CONSULTAR TODOS (READ ALL)
   List<Categoria> categorias = dao.select();

   Ejemplo:
   for (Categoria cat : categorias) {
       System.out.println(cat.getIdCategoria() + " - " + cat.getDescripcion());
   }

4️⃣ CONSULTAR POR ID (READ BY ID)
   Categoria categoria = dao.selectById(1);

   Ejemplo:
   if (categoria != null) {
       System.out.println(categoria.getDescripcion());
   }

5️⃣ BUSCAR POR DESCRIPCIÓN (READ BY DESCRIPTION - LIKE)
   List<Categoria> resultados = dao.selectByDescripcion("palabra");

   Ejemplo:
   // Búsqueda exacta
   List<Categoria> list1 = dao.selectByDescripcion("Electrónica");

   // Búsqueda parcial
   List<Categoria> list2 = dao.selectByDescripcion("Electró");

   // Búsqueda por letra
   List<Categoria> list3 = dao.selectByDescripcion("a");

6️⃣ ACTUALIZAR (UPDATE)
   boolean actualizado = dao.update(1, "Nueva descripción");

   Ejemplo:
   dao.update(1, "Electrónica y Gadgets");
   if (actualizado) {
       System.out.println("Actualizado");
   }

7️⃣ ELIMINAR (DELETE)
   boolean eliminado = dao.delete(1);

   Ejemplo:
   if (dao.delete(1)) {
       System.out.println("Eliminado");
   }

8️⃣ CONTAR TOTAL
   int total = dao.getTotalCategorias();
   System.out.println("Total: " + total);

9️⃣ ELIMINAR TODO (solo pruebas)
   dao.deleteAll();

💻 EJEMPLO SIMPLE DE USO
═════════════════════════════════════════════════════════════════════════════════

import py.com.nikosoft.depositos.boundary.CategoriaDAO;
import py.com.nikosoft.depositos.entities.Categoria;
import java.sql.SQLException;
import java.util.List;

public class MiPrograma {
    public static void main(String[] args) {
        try {
            CategoriaDAO dao = new CategoriaDAO();

            // 1. Crear tabla
            dao.createTable();

            // 2. Agregar categorías
            int id1 = dao.add("Electrónica");
            int id2 = dao.add("Ropa");

            // 3. Ver todas
            List<Categoria> todas = dao.select();
            for (Categoria cat : todas) {
                System.out.println(cat.getId() + " - " + cat.getDescripcion());
            }

            // 4. Buscar por ID
            Categoria cat = dao.selectById(id1);
            System.out.println("Encontrada: " + cat.getDescripcion());

            // 5. Buscar por descripción
            List<Categoria> encontradas = dao.selectByDescripcion("Electró");
            System.out.println("Encontradas: " + encontradas.size());

            // 6. Actualizar
            dao.update(id1, "Electrónica y Gadgets");

            // 7. Eliminar
            dao.delete(id2);

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}
